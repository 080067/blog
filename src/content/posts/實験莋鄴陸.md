#include <stdio.h>

## 第 3 题 · 正负交替的等比数列求和

源文件 `3.cpp`

```c
#include <stdio.h>

int main()
{
    int n;
    double sum = 0.0;
    double first = 1.0;

    printf("请输入n的值： ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        return 1;
    }

    for (int i = 0; i < n; i++) {
        if (i % 2 == 0) {
            sum += first;
        } else {
            sum -= first;
        }
        first /= 2.0;
    }

    printf("前%d项之和： %.3f\n", n, sum);
    return 0;
}
```

## 第 7 题 · 查找回文质数

源文件 `7.cpp`

```c
#include <stdio.h>
#include <math.h>

int n, i, k, h;
int main(){
    printf("请输入数字：");
    scanf("%d",&n);
    if (n < 11)
    {
        return 0;
    }
    else
    for ( i = 11; i < n ; i++)
    {
        h = 1;
        for ( k = 2; k <= sqrt(i); k++)
        {
            if (i % k == 0)
            {
                h = 0;
                break;
            }
        }
        if (h == 1)
        {
            int u = i;
            long long dao = 0;
            while (u > 0)
            {
                dao = dao * 10 + u % 10;
                u /= 10;
            }
            if (dao == i)
            {
                printf("%d是回文质数\n",i);
            }        
        }
    }
}
```

## 第 10 题 · 将偶数拆成两个素数之和

源文件 `10.cpp`

```c
#include <stdio.h>

int zhi_shu(int n)
{
    if (n < 2) {
        return 0;
    }

    for (int k = 2; k <= n / k; k++) {
        if (n % k == 0) {
            return 0;
        }
    }

    return 1;
}

int main(void)
{
    int x;

    printf("请输入x：");
    if (scanf("%d", &x) != 1 || x <= 2 || x % 2 != 0) {
        return 1;
    }

    for (int i = 2; i <= x / 2; i++) {
        int j = x - i;

        if (zhi_shu(i) && zhi_shu(j)) {
            printf("%d=%d+%d\n", x, i, j);
        }
    }

    return 0;
}
```
